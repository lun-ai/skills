# Scaffolding a New Manuscript Pipeline

This guide walks through setting up a reproducible manuscript pipeline from scratch.
The result is a repository where every number, figure, and table traces to source data
and the full paper regenerates from a single `make paper-reproduce` command.

---

## Step 3: Create the Paper Number System

### 3a. The PaperNumber dataclass

Create `paper_artifacts/numbers/paper_number.py`:

```python
"""Paper number data model with full provenance."""
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Optional

@dataclass(frozen=True)
class PaperNumber:
    macro_name: str              # LaTeX macro name, e.g. "ExperimentSampleCount"
    value: object                # Raw value for auditing (int, float, str)
    display_value: str           # LaTeX-ready display string, e.g. "1{,}234"
    unit: str                    # "count", "percent", "percentage_points", etc.
    source_dataset: str          # Which curated file was read
    source_script: str           # Which script computed this
    source_function_or_query: str  # Specific function or aggregation
    upstream_raw_artifacts: str  # Raw files behind the curated input
    manuscript_location: str     # Where this appears in the paper
    description: str             # Human-readable explanation
    operation_summary: str       # Plain-English filter + denominator + arithmetic
    raw_inputs: list[str] = field(default_factory=list)
    derived_inputs: list[str] = field(default_factory=list)
    number_group: str = ""
```

### 3b. The number builder

Create `paper_artifacts/numbers/build.py`:

```python
"""Build paper_numbers.tex, .tsv, and .json from registered number groups."""
from __future__ import annotations
import csv, json, sys
from datetime import datetime, timezone
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[2]
NUMBERS_DIR = REPO_ROOT / "paper" / "numbers_source"
from paper_artifacts.numbers.registry import collect_all_numbers

def build_tex(numbers):
    lines = ["% Auto-generated by paper_artifacts/numbers/build.py"]
    lines.append(r'\providecommand{\PaperNumber}[1]{\csname #1\endcsname}')
    for n in numbers:
        lines.append(
            rf'\expandafter\newcommand\csname {n.macro_name}\endcsname{{{n.display_value}}}'
        )
    return "\n".join(lines) + "\n"

def build_tsv(numbers):
    rows = []
    for n in numbers:
        rows.append({
            "macro_name": n.macro_name,
            "value": n.value,
            "display_value": n.display_value,
            "unit": n.unit,
            "source_dataset": n.source_dataset,
            "source_script": n.source_script,
            "source_function_or_query": n.source_function_or_query,
            "upstream_raw_artifacts": n.upstream_raw_artifacts,
            "manuscript_location": n.manuscript_location,
            "description": n.description,
            "operation_summary": n.operation_summary,
            "raw_inputs": "; ".join(n.raw_inputs),
            "derived_inputs": "; ".join(n.derived_inputs),
            "number_group": n.number_group,
            "generated_at": datetime.now(timezone.utc).isoformat(),
        })
    return rows

def main():
    NUMBERS_DIR.mkdir(parents=True, exist_ok=True)
    numbers = collect_all_numbers()

    (NUMBERS_DIR / "paper_numbers.tex").write_text(build_tex(numbers), encoding="utf-8")

    with (NUMBERS_DIR / "paper_numbers.tsv").open("w", newline="", encoding="utf-8") as f:
        if numbers:
            w = csv.DictWriter(f, fieldnames=build_tsv(numbers)[0].keys(), delimiter="\t")
            w.writeheader()
            w.writerows(build_tsv(numbers))

    with (NUMBERS_DIR / "paper_numbers.json").open("w", encoding="utf-8") as f:
        json.dump(build_tsv(numbers), f, indent=2, default=str)

    print(f"Built {len(numbers)} paper numbers")
    return 0

if __name__ == "__main__":
    sys.exit(main())
```

## Step 4: Create the Publishing Export System

Create `paper/publishing/shared/exports.py`:

```python
"""Contract for final publication TSV exports."""
from __future__ import annotations
import importlib
from collections.abc import Callable, Iterable, Sequence
from dataclasses import dataclass
from pathlib import Path
from types import ModuleType
import pandas as pd
from paper.shared.paths import PAPER, REPO_ROOT

@dataclass(frozen=True)
class PublishingContext:
    """Shared inputs for publication export providers. Extend as needed."""
    pass  # Add project-specific shared DataFrames here

@dataclass(frozen=True)
class PublishingExport:
    """One final TSV handoff artifact."""
    artifact_id: str   # Semantic path, e.g. "figures/main/model_comparison"
    filename: str       # Output filename, must end in .tsv
    frame: pd.DataFrame
    float_format: str | None = None

PublishingProvider = Callable[[PublishingContext], Iterable[PublishingExport]]

def discover_export_modules(source_roots: Sequence[Path] | None = None) -> list[ModuleType]:
    roots = source_roots or (PAPER / "figures", PAPER / "tables")
    modules = []
    for root in roots:
        for make_file in sorted(root.rglob("make.py")):
            if any(part in {"outputs", "tests", "__pycache__"} for part in make_file.parts):
                continue
            rel = make_file.relative_to(REPO_ROOT).with_suffix("")
            modules.append(importlib.import_module(".".join(rel.parts)))
    return modules

def export_providers(modules: Iterable[ModuleType]) -> list[PublishingProvider]:
    return [getattr(m, "publishing_exports") for m in modules
            if hasattr(m, "publishing_exports")]

def collect_exports(providers, context):
    exports, seen = [], set()
    for provider in providers:
        for export in provider(context):
            if export.relative_path in seen:
                raise ValueError(f"Duplicate export: {export.relative_path}")
            seen.add(export.relative_path)
            exports.append(export)
    return exports
```

## Step 5: Create the Makefile

```makefile
# ── Paper Pipeline ──────────────────────────────────────────────

.PHONY: paper-artifacts paper-figures paper-tables paper-numbers \
        paper-publishing-artifacts paper-readable paper-pdf paper-check \
        paper-reproduce

paper-artifacts:
ifdef DATASET
	python paper_artifacts/derived/$(DATASET)/extract.py
else
	@echo "Specify DATASET=<name> or add an 'all' target"
endif

paper-figures:
ifdef FIGURE
	python paper/figures/fig_$(FIGURE)/make.py
else
	@for d in paper/figures/fig_*/make.py; do python "$$d"; done
	@for d in paper/figures/supp/*/make.py; do python "$$d" 2>/dev/null; done
	@for d in paper/figures/analyses/*/make.py; do python "$$d" 2>/dev/null; done
endif

paper-tables:
ifdef TABLE
	python paper/tables/supp/supp_table_$(TABLE)/make.py
else
	@for d in paper/tables/supp/*/make.py; do python "$$d"; done
endif

paper-numbers:
	python paper_artifacts/numbers/build.py

paper-publishing-artifacts:
	python paper/publishing/make.py

paper-readable:
	python scripts/generate_readable.py

paper-pdf: paper-readable
	cd paper && pdflatex -interaction=nonstopmode main_development.tex
	cd paper && bibtex main_development
	cd paper && pdflatex -interaction=nonstopmode main_development.tex
	cd paper && pdflatex -interaction=nonstopmode main_development.tex

paper-check:
	python -m pytest paper/figures/tests/ paper/tables/tests/ paper/publishing/tests/ -v

paper-reproduce: paper-artifacts paper-figures paper-tables paper-numbers \
                 paper-publishing-artifacts paper-pdf paper-check
	@echo "Full paper pipeline complete."
```
